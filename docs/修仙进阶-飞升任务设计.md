# 飞升任务链设计 · 金丹→元婴

## 概述

飞升是凡界到仙界的关键转折点。金丹境玩家需要完成一系列试炼，证明自己有资格进入神秘河（仙界）。

**任务ID范围**: 600100 ~ 600110（自定义）
**前置条件**: 金丹境 + 等级≥140 + 涅槃≥3次
**任务NPC**: 修仙长老（新增NPC，ID待定）
**任务时长**: 约1~2小时（单人可完成）

---

```
    ↓

    ↓

```

## 特殊设计

### 1. 飞升台位置
- **地图**: 时间神殿最深处（地图ID: 240080000 或自定义）
- **NPC**: 飞升台（NPC ID: 自定义）
- **进入条件**: 持有飞升令牌 + 金丹境

### 2. 飞升特效
- 使用现有特效系统
- 播放飞升动画（约5秒）
- 全服广播飞升消息

### 3. 失败机制
- 如果在飞升过程中掉线，任务进度保留
- 重新登录后可以继续飞升
- 凡界精粹不消耗（直到飞升成功）

### 4. 组队机制
- 飞升任务可以组队完成
- 组队击杀BOSS时，所有队员都可以获得材料
- 飞升过程必须单人完成

### 修仙长老NPC脚本框架

```javascript
// 修仙长老NPC脚本
var status = -1;

function start() {
    status = -1;
    action(1, 0, 0);
}

function action(mode, type, selection) {
    if (mode == -1) {
        cm.dispose();
        return;
    }
    
    if (mode == 1) status++;
    else status--;
    
    if (status == 0) {
        // 检查是否金丹境
        var realmId = getPlayerRealm();
        if (realmId < 4) {
            cm.sendOk("你还没有达到金丹境，无法进行飞升试炼。");
            cm.dispose();
            return;
        }
        
        // 检查是否已完成飞升
        if (realmId >= 5) {
            cm.sendOk("你已经完成了飞升，无需再次试炼。");
            cm.dispose();
            return;
        }
        
        // 检查等级和涅槃次数
        var level = cm.getPlayer().getLevel();
        var reborns = cm.getPlayer().getReborns();
        if (level < 140 || reborns < 3) {
            cm.sendOk("飞升需要等级≥140且涅槃≥3次。");
            cm.dispose();
            return;
        }
        
        // 显示飞升任务介绍
        var text = "#e#b修仙长老#k#n\r\n\r\n";
        text += "你已经达到了金丹境，实力在凡界已属顶尖。\r\n";
        text += "但要进入仙界（神秘河），你需要通过飞升试炼。\r\n\r\n";
        text += "飞升试炼需要你证明自己的实力——\r\n";
        text += "击败凡界五大强敌，收集他们的精粹之力。\r\n\r\n";
        text += "#L0#接受飞升试炼#l\r\n";
        text += "#L1#查看飞升进度#l\r\n";
        text += "#L2#了解飞升详情#l\r\n";
        cm.sendSimple(text);
        
    } else if (status == 1) {
        if (selection == 0) {
            // 接受飞升任务
            if (!cm.haveQuest(600100)) {
                cm.startQuest(600100);
                cm.sendOk("飞升试炼已开启！\r\n\r\n" +
                    "你需要完成以下试炼：\r\n" +
                    "1. 击败扎昆（火毒森林）\r\n" +
                    "2. 击败暗黑龙王（龙之巢穴）\r\n" +
                    "3. 击败女皇（女皇城堡）\r\n" +
                    "4. 击败希拉（希拉城堡）\r\n" +
                    "5. 击败粉豆（时间神殿）\r\n\r\n" +
                    "完成后回来找我。");
            } else {
                cm.sendOk("你已经接受了飞升试炼。");
            }
            cm.dispose();
            
        } else if (selection == 1) {
            // 查看飞升进度
            showQuestProgress();
            
        } else if (selection == 2) {
            // 了解飞升详情
            cm.sendOk("飞升试炼详情：\r\n\r\n" +
                "1. 击败凡界五大BOSS\r\n" +
                "2. 收集五种精粹材料\r\n" +
                "3. 合成凡界精粹\r\n" +
                "4. 前往飞升台\r\n" +
                "5. 使用凡界精粹飞升\r\n\r\n" +
                "飞升后你将进入元婴境，解锁神秘河！");
            cm.dispose();
        }
    }
}

function showQuestProgress() {
    var text = "#e#b飞升进度#k#n\r\n\r\n";
    
    // 检查每个BOSS任务完成情况
    var quests = [
        {id: 600101, name: "扎昆", item: "扎昆的火焰之心"},
        {id: 600102, name: "暗黑龙王", item: "龙王的魂鳞"},
        {id: 600103, name: "女皇", item: "女皇的皇冠碎片"},
        {id: 600104, name: "希拉", item: "希拉的黑暗之心"},
        {id: 600105, name: "粉豆", item: "粉豆的混沌之核"}
    ];
    
    for (var i = 0; i < quests.length; i++) {
        var q = quests[i];
        var completed = cm.haveCompleted(q.id);
        var status = completed ? "#g已完成#k" : "#r未完成#k";
        text += (i+1) + ". " + q.name + " - " + status + "\r\n";
    }
    
    text += "\r\n";
    text += "凡界精粹: " + (cm.haveItem(4039999) ? "#g已合成#k" : "#r未合成#k") + "\r\n";
    
    cm.sendOk(text);
    cm.dispose();
}
```

## 飞升台NPC脚本框架

```javascript
// 飞升台NPC脚本
var status = -1;

function start() {
    status = -1;
    action(1, 0, 0);
}

function action(mode, type, selection) {
    if (mode == -1) {
        cm.dispose();
        return;
    }
    
    if (mode == 1) status++;
    else status--;
    
    if (status == 0) {
        // 检查是否金丹境
        var realmId = getPlayerRealm();
        if (realmId != 4) {
            cm.sendOk("只有金丹境修士才能在此飞升。");
            cm.dispose();
            return;
        }
        
        // 检查是否持有凡界精粹
        if (!cm.haveItem(4039999)) {
            cm.sendOk("你没有凡界精粹，无法飞升。\r\n\r\n" +
                "请先完成飞升试炼，收集凡界精粹。");
            cm.dispose();
            return;
        }
        
        // 确认飞升
        cm.sendYesNo("你准备好了吗？\r\n\r\n" +
            "飞升将经历天劫的考验——\r\n" +
            "雷劫、火劫、风劫、心劫。\r\n\r\n" +
            "只有通过所有考验，\r\n" +
            "你才能突破凡界的束缚，\r\n" +
            "进入仙界（神秘河）。\r\n\r\n" +
            "开始飞升？");
            
    } else if (status == 1) {
        if (selection == 1) { // Yes
            // 执行飞升
            performAscension();
        } else {
            cm.sendOk("你退缩了。准备好了再来。");
            cm.dispose();
        }
    }
}

function performAscension() {
    // 1. 播放飞升特效
    cm.getPlayer().sendEffect("飞升特效");
    
    // 2. 全服广播
    cm.getPlayer().sendFullServerBroadcast(
        cm.getPlayer().getName() + " 正在飞升！"
    );
    
    // 3. 等待5秒（特效时间）
    cm.getPlayer().dropMessage(5, "天劫降临...");
    
    // 4. 执行飞升操作
    // 设置境界为元婴
    setPlayerRealm(5);
    
    // 等级重置为140级
    cm.getPlayer().setLevel(140);
    
    // 属性保留（不重置AP）
    
    // 涅槃次数+1
    cm.getPlayer().gainReborns(1);
    
    // 消耗凡界精粹
    cm.gainItem(4039999, -1);
    
    // 5. 发放奖励
    giveAscensionRewards();
    
    // 6. 完成任务
    cm.completeQuest(600108);
    
    // 7. 全服广播
    cm.getPlayer().sendFullServerBroadcast(
        "恭喜 " + cm.getPlayer().getName() + " 飞升成功！进入元婴境！"
    );
    
    cm.sendOk("恭喜你，飞升成功！\r\n\r\n" +
        "你已经突破了凡界的束缚，\r\n" +
        "进入了元婴境。\r\n\r\n" +
        "神秘河的大门已经为你打开！\r\n\r\n" +
        "（请查看背包中的飞升大礼包）");
    
    cm.dispose();
}

function giveAscensionRewards() {
    // 称号
    // cm.gainTitle("元婴修士");
    
    // 偷学技能点
    saveStealKillCount(3);
    
    // 装备强化券
    cm.gainItem(2049115, 15); // +10强化券
    
    // 被动技能
    // cm.teachSkill(元婴护体技能ID);
    
    // 飞升大礼包
    cm.gainItem(2430999, 1);
}
```
